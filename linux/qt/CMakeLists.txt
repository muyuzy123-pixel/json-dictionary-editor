if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Qt GUI and file layer require Linux; core targets remain portable.")
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Test LinguistTools)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

add_library(jsondict_files STATIC file_store.cpp)
target_include_directories(jsondict_files PUBLIC .)
target_link_libraries(jsondict_files PUBLIC jsondict_session Qt6::Core)
target_compile_options(jsondict_files PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(jsondict_file_tests ../tests/file_tests.cpp)
target_link_libraries(jsondict_file_tests PRIVATE jsondict_files)
add_test(NAME linux_atomic_files COMMAND jsondict_file_tests)

# Core-only builds never enter this directory or find/link Qt.

target_link_libraries(jsondict_file_tests PRIVATE ${CMAKE_DL_LIBS})
add_library(jsondict_file_faults SHARED ../tests/file_faults.cpp)
target_link_libraries(jsondict_file_faults PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(jsondict_file_faults PROPERTIES AUTOMOC OFF)
foreach(fault write_short_eintr write_enospc fsync_file_eio fsync_dir_pre_eio fsync_dir_commit_eio fsync_dir_final_eio rename_enosys rename_eopnotsupp rename_einval rename_eio create_eacces)
 add_test(NAME linux_fault_${fault} COMMAND ${CMAKE_COMMAND} -E env
 "LD_PRELOAD=$<TARGET_FILE:jsondict_file_faults>" "JDE_TEST_FAULT=${fault}" $<TARGET_FILE:jsondict_file_tests> --fault)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/prepare_resources.py" RESULT_VARIABLE JDE_RESOURCE_RESULT)
if(NOT JDE_RESOURCE_RESULT EQUAL 0)
 message(FATAL_ERROR "Original icon resource preparation failed")
endif()
qt_add_translation(JDE_QM_FILES ../translations/editor_en.ts ../translations/editor_zh_CN.ts)
add_custom_target(jsondict_translations DEPENDS ${JDE_QM_FILES})
configure_file(translations.qrc.in "${CMAKE_CURRENT_BINARY_DIR}/translations.qrc" @ONLY)
add_library(jsondict_widgets STATIC language.cpp tree_model.cpp editor_window.cpp raw_dialog.cpp "${CMAKE_CURRENT_BINARY_DIR}/translations.qrc" sample.qrc)
add_dependencies(jsondict_widgets jsondict_translations)
target_link_libraries(jsondict_widgets PUBLIC jsondict_files Qt6::Widgets)
target_include_directories(jsondict_widgets PUBLIC .)
target_compile_options(jsondict_widgets PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(json-dictionary-editor main.cpp)
target_link_libraries(json-dictionary-editor PRIVATE jsondict_widgets)
add_executable(jsondict_interaction_tests ../tests/interaction_tests.cpp)
target_link_libraries(jsondict_interaction_tests PRIVATE jsondict_widgets Qt6::Test)
add_test(NAME linux_qt_interactions COMMAND jsondict_interaction_tests)
set_tests_properties(linux_qt_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
add_executable(jsondict_save_state_tests ../tests/save_state_tests.cpp)
target_link_libraries(jsondict_save_state_tests PRIVATE jsondict_widgets Qt6::Test ${CMAKE_DL_LIBS})
add_test(NAME linux_qt_save_failure_state COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:jsondict_file_faults>" "QT_QPA_PLATFORM=offscreen" $<TARGET_FILE:jsondict_save_state_tests> -v2)
set_tests_properties(linux_qt_save_failure_state PROPERTIES TIMEOUT 120)
install(TARGETS json-dictionary-editor RUNTIME DESTINATION bin)
install(FILES ../resources/json-dictionary-editor.png DESTINATION share/icons/hicolor/256x256/apps)
install(FILES ../packaging/json-dictionary-editor.desktop DESTINATION share/applications)
install(FILES ../../LICENSE DESTINATION share/doc/json-dictionary-editor RENAME copyright)
install(FILES ../README.md ../../windows/resources/SampleDictionary.json DESTINATION share/doc/json-dictionary-editor)
add_executable(jsondict_metadata_race_tests ../tests/metadata_race_tests.cpp)
target_link_libraries(jsondict_metadata_race_tests PRIVATE jsondict_files)
foreach(case target_xattr_before_rename temporary_xattr_before_rename target_xattr_at_commit temporary_xattr_at_commit target_xattr_change_at_commit temporary_xattr_change_at_commit target_acl_at_commit temporary_acl_at_commit target_xattr_after_commit backup_acl_after_commit default_acl_temporary target_xattr_process_at_commit target_acl_process_at_commit)
 add_test(NAME linux_metadata_${case} COMMAND jsondict_metadata_race_tests ${case})
 set_tests_properties(linux_metadata_${case} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
